    <script>
        function generateVideo() {
            let promptText = document.getElementById("prompt").value;
            let outputDiv = document.getElementById("output");
            
            if(!promptText) {
                alert("Pehle prompt likhein!");
                return;
            }

            outputDiv.innerHTML = "<p>Video ban rahi hai, thora intezar karein...</p>";

            // Pollinations Open-Source Unlimited Engine URL
            let videoUrl = "https://image.pollinations.ai/prompt/" + encodeURIComponent(promptText) + "?model=flux&nologo=true";

            // Render output
            outputDiv.innerHTML = `<img src="${videoUrl}" id="videoResult" alt="Generated Content"><br><br>
                                   <a href="${videoUrl}" download="ai_video.jpg" style="color:#00ffcc;">Download Karein</a>`;
        }
    </script>
